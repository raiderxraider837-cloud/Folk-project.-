local function DetectWall()  
	if not Root then  
		return false, nil  
	end  
  
	local params = GetRayParams()  
  
	local directions = {  
		Vector3.new(1, 0, 0),  
		Vector3.new(-1, 0, 0),  
		Vector3.new(0, 0, 1),  
		Vector3.new(0, 0, -1)  
	}  
  
	local closestNormal = nil  
	local closestDistance = math.huge  
  
	for _, direction in ipairs(directions) do  
		local result = workspace:Raycast(  
			Root.Position,  
			direction * WALL_DISTANCE,  
			params  
		)  
  
		if result then  
			local normal = Vector3.new(  
				result.Normal.X,  
				0,  
				result.Normal.Z  
			)  
  
			if normal.Magnitude > 0.2 then  
				local distance = (result.Position - Root.Position).Magnitude  
  
				if distance < closestDistance then  
					closestDistance = distance  
					closestNormal = normal.Unit  
				end  
			end  
		end  
	end  
  
	if closestNormal then  
		return true, closestNormal  
	end  
  
	return false, nil  
end  
  
local function UpdateWallContact()

	WallContact, WallNormal = DetectWall()

local function DetectCeiling()  
	if not Head then  
		return false  
	end  
  
	local params = GetRayParams()  
  
	local offsets = {  
		Vector3.new(0, 0, 0),  
		Vector3.new(0.28, 0, 0),  
		Vector3.new(-0.28, 0, 0)  
	}  
  
	local hits = 0  
  
	for _, offset in ipairs(offsets) do  
		local origin = Head.Position  
			+ Vector3.new(0, HEAD_CHECK_HEIGHT, 0)  
			+ offset  
  
		local result = workspace:Raycast(  
			origin,  
			Vector3.new(0, HEAD_CHECK_DISTANCE, 0),  
			params  
		)  
  
		if result  
			and result.Instance  
			and not result.Instance:IsA("TrussPart")  
			and result.Normal.Y <= HEAD_NORMAL_THRESHOLD then  
			hits += 1  
		end  
	end  
  
	return hits >= 2  
end  
  
local function UpdateCeilingLock()  
	if not CeilingLock or not Root then  
		return  
	end  
  
	local now = os.clock()  
	local touchingCeiling = DetectCeiling()  
	local velocityY = Root.AssemblyLinearVelocity.Y  
  
	if touchingCeiling then  
		CeilingClearSince = 0  
		return  
	end  
  
	if now - CeilingLockStarted < CEILING_LOCK_MIN_TIME then  
		CeilingClearSince = 0  
		return  
	end  
  
	if velocityY > 0.05 then  
		CeilingClearSince = 0  
		return  
	end  
  
	if CeilingClearSince == 0 then  
		CeilingClearSince = now  
		return  
	end  
  
	if now - CeilingClearSince >= CEILING_CLEAR_TIME then  
		CeilingLock = false  
		CeilingLockStarted = 0  
		CeilingClearSince = 0  
	end  
end  
  
local function ResolveCeiling()  
	if not Root or not Humanoid then  
		return false

	if velocity.Y <= 0.05 or not DetectCeiling() then  
		return false

	CeilingLock = true  
	CeilingLockStarted = os.clock()  
	CeilingClearSince = 0  
  
	JumpActive = false  
	CustomJumpActive = false  
	JumpVelocity = 0  
	JumpFlightTime = 0  
  
	Humanoid.Jump = false  
  
	StopTrack(JumpTrack, 0.1)  
  
	return true  
end

local function StartJump()  
	if not Humanoid or not Root or CeilingLock then  
		return false  
	end  
  
	local now = os.clock()  
  
	if now - LastJumpRequest < JUMP_REQUEST_COOLDOWN then  
		return false  
	end  
  
	local state = Humanoid:GetState()  
	local success  
  
	if state == Enum.HumanoidStateType.Climbing then  
		success = BeginTrussJump()  
	else  
		success = BeginCustomJump()  
	end  
  
	if success then  
		LastJumpRequest = now  
	end  
  
	return success  
end  
  
local function StartWallhop()  
	if not Humanoid or not Root or CeilingLock then  
		return false  
	end  
  
	if not WallContact then  
		return false  
	end  
  
	local now = os.clock()  
  
	if now - LastWallhop < WALLHOP_COOLDOWN then  
		return false  
	end  
  
	local success = BeginCustomJump()  
  
	if success then  
		LastWallhop = now  
		LastJumpRequest = now  
	end  
  
	return success  
end  
  
local function UpdateJumpState()  
	if not JumpActive or not Humanoid or not Root then  
		return  
	end  
  
	if CeilingLock then  
		JumpActive = false  
		CustomJumpActive = false  
		return  
	end  
  
	local now = os.clock()  
	local state = Humanoid:GetState()  
  
	if state == Enum.HumanoidStateType.Climbing then  
		JumpActive = false  
		CustomJumpActive = false  
		JumpVelocity = 0  
		JumpFlightTime = 0  
		return  
	end  
  
	if IsGrounded(state) and now - JumpStartTime > 0.08 then  
		JumpActive = false  
		CustomJumpActive = false  
		JumpVelocity = 0  
		JumpFlightTime = 0  
		return  
	end  
  
	if now - JumpStartTime >= JumpFlightTime then  
		JumpActive = false  
		CustomJumpActive = false  
		JumpVelocity = 0  
		JumpFlightTime = 0  
	end  
end

-- MOVEMENT CALCULATION  
  
local function CalculateMovement(position, velocity, previousPosition)  
    local delta = position - (previousPosition or position)  
  
    Stats.Position = position  
    Stats.Velocity = velocity  
  
    Stats.HorizontalSpeed = Vector3.new(  
        velocity.X,  
        0,  
        velocity.Z  
    ).Magnitude  
  
    Stats.VerticalSpeed = velocity.Y  
  
    return delta  
end  
  
-- TAKEOFF DETECTION  
  
local function DetectTakeoff(velocity)  
    if not PendingJump or not Humanoid then  
        return  
    end  
  
    local elapsed = Frame - PendingJump.inputFrame  
    local state = Humanoid:GetState()  
  
    local jumping =  
        state == Enum.HumanoidStateType.Jumping  
  
    local upwardImpulse =  
        velocity.Y > Config.VerticalThreshold  
  
    if jumping or upwardImpulse then  
        Stats.Jumps += 1  
        Stats.LastJumpDelay = elapsed  
  
        ActiveJump = {  
            inputFrame = PendingJump.inputFrame,  
            takeoffFrame = Frame,  
            startTime = os.clock(),  
            delayFrames = elapsed,  
            startPosition = Root.Position,  
            history = {}  
        }  
  
        Log(  
            "TAKEOFF | FRAME: "  
            .. Frame  
            .. " | DELAY: "  
            .. elapsed,  
            Color3.fromRGB(80, 220, 150)  
        )  
  
        PendingJump = nil  
  
    elseif elapsed > Config.MaxInputFrames then  
        Stats.Missed += 1  
  
        Log(  
            "MISSED INPUT | FRAME: " .. Frame,  
            Color3.fromRGB(255, 100, 100)  
        )  
  
        PendingJump = nil  
    end  
end  
  
-- JUMP HISTORY  
  
local function UpdateJumpHistory(delta, velocity)  
    if not ActiveJump then  
        return  
    end  
  
    table.insert(ActiveJump.history, {  
        frame = Frame,  
  
        x = delta.X,  
        y = delta.Y,  
        z = delta.Z,  
  
        velocityX = velocity.X,  
        velocityY = velocity.Y,  
        velocityZ = velocity.Z,  
  
        horizontalSpeed = Stats.HorizontalSpeed  
    })  
  
    if #ActiveJump.history > 3000 then  
        table.remove(ActiveJump.history, 1)  
    end  
end  
  
-- LANDING DETECTION  
  
local function DetectLanding()  
    if not ActiveJump or not Humanoid then  
        return  
    end  
  
    if Humanoid.FloorMaterial == Enum.Material.Air then  
        return  
    end  
  
    if Frame <= ActiveJump.takeoffFrame + 1 then  
        return  
    end  
  
    local duration = Frame - ActiveJump.takeoffFrame  
  
    Stats.Landings += 1  
    Stats.LastJumpDuration = duration  
  
    local horizontalDistance = Vector3.new(  
        Root.Position.X - ActiveJump.startPosition.X,  
        0,  
        Root.Position.Z - ActiveJump.startPosition.Z  
    ).Magnitude  
  
    Log(  
        "LANDING | FRAMES: "  
        .. duration  
        .. " | DISTANCE: "  
        .. FormatNumber(horizontalDistance),  
        Color3.fromRGB(100, 210, 160)  
    )  
  
    ActiveJump = nil  
end  
  
-- TAS RECORDING  
  
local function RecordFrame(position, velocity)  
    if not Recorder.Recording then  
        return  
    end  
  
    if #Recorder.Frames >= Config.MaxRecordedFrames then  
        Recorder.Recording = false  
        Recorder.Enabled = false  
  
        Log("TAS MEMORY LIMIT REACHED")  
        return  
    end  
  
    if not Humanoid then  
        return  
    end  
  
    local state = Humanoid:GetState()  
    local floor = Humanoid.FloorMaterial  
    local moveDirection = Humanoid.MoveDirection  
  
    table.insert(Recorder.Frames, {  
        frame = Frame,  
  
        time = os.clock() - Recorder.StartTime,  
  
        position = {  
            X = position.X,  
            Y = position.Y,  
            Z = position.Z  
        },  
  
        velocity = {  
            X = velocity.X,  
            Y = velocity.Y,  
            Z = velocity.Z  
        },  
  
        moveDirection = {  
            X = moveDirection.X,  
            Y = moveDirection.Y,  
            Z = moveDirection.Z  
        },  
  
        state = state.Name,  
        floor = floor.Name,  
  
        fps = Stats.FPS  
    })  
end  
  
-- PHYSICS TRACKER  
  
local FPSHistory = {}  
  
RunService.Heartbeat:Connect(function(dt)  
    Frame += 1  
    Stats.Frames = Frame  
  
    if dt > 0 then  
        Stats.FPS = 1 / dt  
  
        table.insert(FPSHistory, Stats.FPS)  
  
        while #FPSHistory > Config.AverageFPSWindow do  
            table.remove(FPSHistory, 1)  
        end  
  
        local total = 0  
  
        for _, sample in ipairs(FPSHistory) do  
            total += sample  
        end  
  
        if #FPSHistory > 0 then  
            Stats.AverageFPS = total / #FPSHistory  
        end  
    end  
  
    if not Character  
        or not Humanoid  
        or not Root  
        or not Root.Parent  
        or Humanoid.Health <= 0 then  
  
        return  
    end  
  
    local position = Root.Position  
    local velocity = Root.AssemblyLinearVelocity  
  
    local delta = CalculateMovement(  
        position,  
        velocity,  
        PreviousPosition  
    )  
  
    if Config.Enabled then  
        DetectTakeoff(velocity)  
        UpdateJumpHistory(delta, velocity)  
        DetectLanding()  
    end  
  
    RecordFrame(position, velocity)  
  
    PreviousPosition = position  
    PreviousVelocity = velocity  
end)

local function RecordFrame(deltaTime)
    if not Recorder.Enabled or not Recorder.Recording then
        return
    end

    if #Recorder.Frames >= Config.MaxRecordedFrames then
        Recorder.Recording = false
        Recorder.Enabled = false

        Log("TAS RECORDING LIMIT REACHED")
        return
    end

    if not Root then
        return
    end

    local position = Root.Position
    local velocity = Root.AssemblyLinearVelocity

    table.insert(Recorder.Frames, {
        t = os.clock() - Recorder.StartTime,
        dt = deltaTime,

        p = {
            position.X,
            position.Y,
            position.Z
        },

        v = {
            velocity.X,
            velocity.Y,
            velocity.Z
        },

        grounded = IsGrounded()
    })
end

local function StartJump()
    if not PendingJump or not Root or not Humanoid then
        return
    end

    local velocity = Root.AssemblyLinearVelocity

    ActiveJump = {
        startTime = os.clock(),
        startFrame = Frame,
        startPosition = Root.Position,
        startVelocity = velocity
    }

    Stats.Jumps += 1

    Stats.LastJumpDelay = os.clock() - PendingJump.time

    Log(
        "JUMP DETECTED | VY: "
        .. FormatNumber(velocity.Y)
    )

    PendingJump = nil
end

local function FinishJump()
    if not ActiveJump then
        return
    end

    Stats.Landings += 1

    Stats.LastJumpDuration =
        os.clock() - ActiveJump.startTime

    Log(
        "LANDING | DURATION: "
        .. FormatNumber(Stats.LastJumpDuration)
        .. "s"
    )

    ActiveJump = nil
end
