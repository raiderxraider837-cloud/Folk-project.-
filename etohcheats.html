local function GetJumpVelocity()
	return math.sqrt(2 * workspace.Gravity * JUMP_HEIGHT)
end

local function GetJumpFlightTime()
	return 2 * GetJumpVelocity() / workspace.Gravity
end

local function StopTrack(track, fadeTime)
	if track and track.IsPlaying then
		track:Stop(fadeTime or 0.15)
	end
end

local function GetRayParams()
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {Character}
	params.IgnoreWater = true
	params.RespectCanCollide = true
	return params
end

local function IsGrounded(state)
	return state == Enum.HumanoidStateType.Running
		or state == Enum.HumanoidStateType.RunningNoPhysics
		or state == Enum.HumanoidStateType.Landed
end

local function LoadAnimations()
	if not Animator then
		return
	end

	if JumpTrack then
		JumpTrack:Destroy()
	end

	if FallTrack then
		FallTrack:Destroy()
	end

	if ClimbTrack then
		ClimbTrack:Destroy()
	end

	local jumpAnimation = Instance.new("Animation")
	jumpAnimation.AnimationId = JUMP_ANIMATION_ID

	local fallAnimation = Instance.new("Animation")
	fallAnimation.AnimationId = FALL_ANIMATION_ID

	local climbAnimation = Instance.new("Animation")
	climbAnimation.AnimationId = CLIMB_ANIMATION_ID

	JumpTrack = Animator:LoadAnimation(jumpAnimation)
	FallTrack = Animator:LoadAnimation(fallAnimation)
	ClimbTrack = Animator:LoadAnimation(climbAnimation)

	JumpTrack.Priority = Enum.AnimationPriority.Action
	FallTrack.Priority = Enum.AnimationPriority.Action
	ClimbTrack.Priority = Enum.AnimationPriority.Action

	JumpTrack.Looped = false
	FallTrack.Looped = false
	ClimbTrack.Looped = true

	JumpTrack:AdjustSpeed(JUMP_ANIMATION_SPEED)
	FallTrack:AdjustSpeed(FALL_ANIMATION_SPEED)
	ClimbTrack:AdjustSpeed(CLIMB_ANIMATION_SPEED)
end

local function PlayJumpAnimation()
	if not JumpTrack or JumpAnimationPlayed then
		return
	end

	JumpAnimationPlayed = true
	FallAnimationPlayed = false
	JumpAnimationStart = os.clock()

	StopTrack(FallTrack, 0.15)
	StopTrack(ClimbTrack, 0.15)

	JumpTrack.TimePosition = 0
	JumpTrack:Play(0.1, 1, JUMP_ANIMATION_SPEED)
end

local function PlayFallAnimation()
	if not FallTrack or FallAnimationPlayed then
		return
	end

	if os.clock() - JumpAnimationStart < JUMP_MIN_ANIMATION_TIME then
		return
	end

	FallAnimationPlayed = true

	StopTrack(JumpTrack, FALL_TRANSITION_TIME)
	StopTrack(ClimbTrack, 0.15)

	FallTrack.TimePosition = 0
	FallTrack:Play(
		FALL_TRANSITION_TIME,
		1,
		FALL_ANIMATION_SPEED
	)
end

local function UpdateAnimations()
	if not Humanoid or not Root then
		return
	end

	local state = Humanoid:GetState()

	if state == Enum.HumanoidStateType.Climbing then
		StopTrack(JumpTrack, 0.15)
		StopTrack(FallTrack, 0.15)

		if ClimbTrack then
			if not ClimbTrack.IsPlaying then
				ClimbTrack.TimePosition = 0
				ClimbTrack:Play(0.15, 1, CLIMB_ANIMATION_SPEED)
			end

			if math.abs(Root.AssemblyLinearVelocity.Y) < 0.05 then
				ClimbTrack:AdjustSpeed(0)
			else
				ClimbTrack:AdjustSpeed(CLIMB_ANIMATION_SPEED)
			end
		end

		return
	end

	StopTrack(ClimbTrack, 0.15)

	if not CustomJumpActive or CeilingLock then
		return
	end

	if not JumpAnimationPlayed then
		PlayJumpAnimation()
	end

	if Root.AssemblyLinearVelocity.Y <= FALL_START_VELOCITY then
		PlayFallAnimation()
	end
end

local function DetectWall()
	if not Root then
		return false, nil
	end

	local params = GetRayParams()

	local directions = {
		Vector3.new(1, 0, 0),
		Vector3.new(-1, 0, 0),
		Vector3.new(0, 0, 1),
		Vector3.new(0, 0, -1)
	}

	local closestNormal = nil
	local closestDistance = math.huge

	for _, direction in ipairs(directions) do
		local result = workspace:Raycast(
			Root.Position,
			direction * WALL_DISTANCE,
			params
		)

		if result then
			local normal = Vector3.new(
				result.Normal.X,
				0,
				result.Normal.Z
			)

			if normal.Magnitude > 0.2 then
				local distance = (result.Position - Root.Position).Magnitude

				if distance < closestDistance then
					closestDistance = distance
					closestNormal = normal.Unit
				end
			end
		end
	end

	if closestNormal then
		return true, closestNormal
	end

	return false, nil
end

local function UpdateWallContact()
	PreviousWallContact = WallContact
	WallContact, WallNormal = DetectWall()
end

local function DetectCeiling()
	if not Head then
		return false
	end

	local params = GetRayParams()

	local offsets = {
		Vector3.new(0, 0, 0),
		Vector3.new(0.28, 0, 0),
		Vector3.new(-0.28, 0, 0)
	}

	local hits = 0

	for _, offset in ipairs(offsets) do
		local origin = Head.Position
			+ Vector3.new(0, HEAD_CHECK_HEIGHT, 0)
			+ offset

		local result = workspace:Raycast(
			origin,
			Vector3.new(0, HEAD_CHECK_DISTANCE, 0),
			params
		)

		if result
			and result.Instance
			and not result.Instance:IsA("TrussPart")
			and result.Normal.Y <= HEAD_NORMAL_THRESHOLD then
			hits += 1
		end
	end

	return hits >= 2
end

local function UpdateCeilingLock()
	if not CeilingLock or not Root then
		return
	end

	local now = os.clock()
	local touchingCeiling = DetectCeiling()
	local velocityY = Root.AssemblyLinearVelocity.Y

	if touchingCeiling then
		CeilingClearSince = 0
		return
	end

	if now - CeilingLockStarted < CEILING_LOCK_MIN_TIME then
		CeilingClearSince = 0
		return
	end

	if velocityY > 0.05 then
		CeilingClearSince = 0
		return
	end

	if CeilingClearSince == 0 then
		CeilingClearSince = now
		return
	end

	if now - CeilingClearSince >= CEILING_CLEAR_TIME then
		CeilingLock = false
		CeilingLockStarted = 0
		CeilingClearSince = 0
	end
end

local function ResolveCeiling()
	if not Root or not Humanoid then
		return false
	end

	if CeilingLock then
		UpdateCeilingLock()
		return true
	end

	if not CustomJumpActive or not JumpActive then
		return false
	end

	if Humanoid:GetState() == Enum.HumanoidStateType.Climbing then
		return false
	end

	local velocity = Root.AssemblyLinearVelocity

	if velocity.Y <= 0.05 or not DetectCeiling() then
		return false
	end

	CeilingLock = true
	CeilingLockStarted = os.clock()
	CeilingClearSince = 0

	JumpActive = false
	CustomJumpActive = false
	JumpVelocity = 0
	JumpFlightTime = 0

	Humanoid.Jump = false

	StopTrack(JumpTrack, 0.1)

	return true
end

local function BeginTrussJump()
	if not Humanoid or not Root or CeilingLock then
		return false
	end

	JumpActive = false
	CustomJumpActive = false
	TrussLaunchActive = true

	JumpVelocity = 0
	JumpFlightTime = 0

	Humanoid.Jump = true

	StopTrack(ClimbTrack, 0.05)

	JumpAnimationPlayed = false
	FallAnimationPlayed = false
	JumpAnimationStart = os.clock()

	if JumpTrack then
		JumpTrack.TimePosition = 0
		JumpTrack:Play(0.05, 1, JUMP_ANIMATION_SPEED)
	end

	return true
end

local function BeginCustomJump()
	if not Humanoid or not Root or CeilingLock then
		return false
	end

	JumpVelocity = GetJumpVelocity()
	JumpFlightTime = GetJumpFlightTime()
	JumpStartTime = os.clock()

	local velocity = Root.AssemblyLinearVelocity

	Root.AssemblyLinearVelocity = Vector3.new(
		velocity.X,
		JumpVelocity,
		velocity.Z
	)

	JumpActive = true
	CustomJumpActive = true
	TrussLaunchActive = false

	JumpAnimationPlayed = false
	FallAnimationPlayed = false

	PlayJumpAnimation()

	return true
end

local function StartJump()
	if not Humanoid or not Root or CeilingLock then
		return false
	end

	local now = os.clock()

	if now - LastJumpRequest < JUMP_REQUEST_COOLDOWN then
		return false
	end

	local state = Humanoid:GetState()
	local success

	if state == Enum.HumanoidStateType.Climbing then
		success = BeginTrussJump()
	else
		success = BeginCustomJump()
	end

	if success then
		LastJumpRequest = now
	end

	return success
end

local function StartWallhop()
	if not Humanoid or not Root or CeilingLock then
		return false
	end

	if not WallContact then
		return false
	end

	local now = os.clock()

	if now - LastWallhop < WALLHOP_COOLDOWN then
		return false
	end

	local success = BeginCustomJump()

	if success then
		LastWallhop = now
		LastJumpRequest = now
	end

	return success
end

local function UpdateJumpState()
	if not JumpActive or not Humanoid or not Root then
		return
	end

	if CeilingLock then
		JumpActive = false
		CustomJumpActive = false
		return
	end

	local now = os.clock()
	local state = Humanoid:GetState()

    if state == Enum.HumanoidStateType.Climbing then
		JumpActive = false
		CustomJumpActive = false
		JumpVelocity = 0
		JumpFlightTime = 0
		return
	end

	if IsGrounded(state) and now - JumpStartTime > 0.08 then
		JumpActive = false
		CustomJumpActive = false
		JumpVelocity = 0
		JumpFlightTime = 0
		return
	end

	if now - JumpStartTime >= JumpFlightTime then
		JumpActive = false
		CustomJumpActive = false
		JumpVelocity = 0
		JumpFlightTime = 0
	end
end

	TrussLaunchActive = false

	JumpStartTime = 0
	JumpVelocity = 0
	JumpFlightTime = 0

	WallContact = false
	WallNormal = nil
	PreviousWallContact = false

	CeilingLock = false
	CeilingLockStarted = 0
	CeilingClearSince = 0

	JumpAnimationPlayed = false
	FallAnimationPlayed = false
	JumpAnimationStart = os.clock()

	if StateConnection then
		StateConnection:Disconnect()
		StateConnection = nil
	end

	LoadAnimations()

	StateConnection = Humanoid.StateChanged:Connect(function(_, newState)
		if IsGrounded(newState) then
			LastGrounded = os.clock()

			JumpActive = false
			CustomJumpActive = false

			JumpVelocity = 0
			JumpFlightTime = 0

			JumpAnimationPlayed = false
			FallAnimationPlayed = false

			StopTrack(JumpTrack, 0.2)
			StopTrack(FallTrack, 0.2)
		end

		if newState == Enum.HumanoidStateType.Climbing then
			TrussLaunchActive = false
			StopTrack(JumpTrack, 0.1)
			StopTrack(FallTrack, 0.1)
		elseif newState == Enum.HumanoidStateType.Freefall
			or newState == Enum.HumanoidStateType.Jumping then
			TrussLaunchActive = false
		end
	end)
end

	if not InfiniteJump or not Humanoid or not Root then
		return
	end

	if Humanoid.Health <= 0 or CeilingLock then
		return
	end

	local now = os.clock()

	if JumpInputLocked then
		if now - LastJumpInput < JUMP_INPUT_RELEASE_TIME then
			return
		end

		JumpInputLocked = false
	end

	JumpInputLocked = true
	LastJumpInput = now

	local state = Humanoid:GetState()

	if state == Enum.HumanoidStateType.Climbing then
		StartJump()
		return
	end

	if IsGrounded(state) or now - LastGrounded <= COYOTE_TIME then
		StartJump()
		return
	end

	if WallContact and not PreviousWallContact then
		if StartWallhop() then
			return
		end
	end

	StartJump()
end)

		LastGrounded = os.clock()
	end

	UpdateAnimations()
